# ==== Working with /test directory ====

cd /test

# list parent
ls ..

# ==== wc tests ====
# file
wc /test/skip_test.txt
# file used in uniq ignore-case tests
wc /test/ignore_case.txt
# attempt to wc a directory -> should show error
wc /test

# ==== uniq tests: many modes and combinations ====
# basic adjacent filtering
uniq /test/skip_test.txt
# prefix counts for adjacent groups
uniq -c /test/skip_test.txt
# show only duplicated adjacent groups
uniq -d /test/skip_test.txt
# show only unique adjacent groups
uniq -u /test/skip_test.txt
# global mode: compare all lines ignoring adjacency
uniq -A /test/skip_test.txt
# case-insensitive comparison
uniq -i /test/ignore_case.txt
# ignore first 1 field
uniq -f 1 /test/skip_test.txt
# skip first 1 character
uniq -s 1 /test/skip_test.txt
# combine field and char skips
uniq -f 1 -s 2 /test/skip_test.txt

# extra uniq scenarios (files with different formats)
uniq -i /test/for_i.txt
uniq -c /test/for_c.txt
uniq -f 1 /test/for_f.txt
uniq -s 1 /test/for_s.txt
uniq -f 1 -s 2 /test/for_f.txt

# attempt to uniq a directory (edge case) -> should print error but continue
uniq /test

# ==== error edge cases for reading files ====
wc /nonexistent.txt
uniq /nonexistent.txt
